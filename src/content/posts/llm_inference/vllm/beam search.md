`[STREAMING_PAD]`、`[STREAMING_WORD]` 不应该暴露给用户,  Voxtral Realtime 使用 Beam Search 转录时，最终解码没有传递 `skip_special_tokens`，导致 `[STREAMING_PAD]`、`[STREAMING_WORD]` 等内部控制 token 泄漏到返回文本中。

本修改为 `BeamSearchParams` 增加 `skip_special_tokens`，并在 Online 和 Offline Beam Search 最终解码时使用它。默认过滤特殊 token，同时允许用户显式设置 `False` 保留原始输出。该修改只影响文本解码，不改变 Beam 搜索结果、token IDs 或 logits。



offline test

```python
import os
from pathlib import Path

from mistral_common.protocol.transcription.request import (
    StreamingMode,
    TranscriptionRequest,
)
from mistral_common.tokens.tokenizers.audio import Audio
from mistral_common.tokens.tokenizers.mistral import MistralTokenizer

from vllm import LLM
from vllm.sampling_params import BeamSearchParams


MODEL = "mistralai/Voxtral-Mini-4B-Realtime-2602"
audio_path = Path(os.environ["AUDIO_FILE"])

if not audio_path.is_file():
    raise FileNotFoundError(audio_path)

tokenizer = MistralTokenizer.from_hf_hub(MODEL)
audio = Audio.from_file(str(audio_path), strict=False)

request = TranscriptionRequest(
    audio=audio.to_base64(audio.format),
    streaming=StreamingMode.OFFLINE,
    language="en",
)
tokenized = tokenizer.instruct_tokenizer.encode_transcription(request)

prompt = {
    "prompt_token_ids": tokenized.tokens,
    "multi_modal_data": {
        "audio": [(tokenized.audios[0].audio_array, None)],
    },
}

llm = LLM(
    model=MODEL,
    tokenizer_mode="mistral",
    config_format="mistral",
    load_format="mistral",
    trust_remote_code=True,
    max_model_len=8192,
    max_num_seqs=1,
    gpu_memory_utilization=0.92,
    enforce_eager=True,
)

params = BeamSearchParams(
    beam_width=2,
    max_tokens=256,
    temperature=0.0,
)

result = llm.beam_search(
    [prompt],
    params,
    use_tqdm=True,
)[0]

print("beam count:", len(result.sequences))

for index, sequence in enumerate(result.sequences):
    text = sequence.text or ""

    print()
    print("beam:", index)
    print("finish_reason:", sequence.finish_reason)
    print("token_count:", len(sequence.tokens))
    print("STREAMING_PAD count:", text.count("[STREAMING_PAD]"))
    print("STREAMING_WORD count:", text.count("[STREAMING_WORD]"))
    print("text:", repr(text))
```
